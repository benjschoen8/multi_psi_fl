\subsection*{Preliminaries: shared subroutines}
\label{sec:sub}
\noindent Small helpers reused by the algorithms below, stated once here
rather than repeated in each one.

\small
\begin{algorithmic}[1]

\Procedure{Unit}{$V$}
  \Comment{$V$ is a matrix; rows are scaled to length 1}
  \For{each row $v$ of $V$}
    \State $v \gets v / \lVert v \rVert$
      \Comment{comparison is then by angle, so no calibration or common scale
               is needed across clients}
  \EndFor
  \State \Return $V$
\EndProcedure

\Statex

\Procedure{OneSided}{$R$}
  \Comment{$R[a,b]$ is a ladder rank; returns what ONE client can emit from
           its OWN rows, knowing nothing of the peer's preferences}
  \State $M \gets \emptyset$
  \For{each row index $a$ of $R$}
    \State $m \gets \max_b R[a,b]$
    \If{$m = 0$}
      \State \textbf{continue}
        \Comment{never cleared the floor: no candidate at any rung}
    \EndIf
    \If{$\big|\{\,b : R[a,b] = m\,\}\big| > 1$}
      \State \textbf{continue}
        \Comment{tie -- drop it; costs recall, never precision}
    \EndIf
    \State $M \gets M \cup \{(a,\, \arg\max_b R[a,b])\}$
  \EndFor
  \State \Return $M$
\EndProcedure

\Statex

\Procedure{Mutual}{$R$}
  \Comment{keeps a pair only if each side is the other's top choice}
  \State $V_i \gets \onesided(R)$,\quad $V_j \gets \onesided(R^{\top})$
  \State \Return $\{\,(a,b) \in V_i : (b,a) \in V_j\,\}$
\EndProcedure

\Statex

\Procedure{AttnCoord}{$q_a,\, K,\, V$}
  \Comment{cross-attention pooling of class $a$'s query over the PUBLIC,
           shared anchor keys/values $K=\{k_p\},V=\{v_p\}$; same $K,V$ used by
           every client, so the output coordinate lands in one shared space}
  \For{each anchor $p$}
    \State $\mathrm{score}_p \gets \dfrac{q_a \cdot k_p}{\sqrt{d}}$
  \EndFor
  \State $\alpha \gets \mathrm{softmax}(\mathrm{score})$
    \Comment{attention weights over the anchor set -- itself usable as a
             coordinate vector, one entry per anchor}
  \State $\mathrm{coord} \gets \sum_p \alpha_p\, v_p$
    \Comment{pooled embedding -- an alternative coordinate, dimension set by
             $v_p$ rather than by $|A|$}
  \State \Return $(\alpha,\ \mathrm{coord})$
\EndProcedure

\end{algorithmic}
\normalsize

\begin{algorithm}[p]
\caption{Shared-vocabulary description PSI}
\label{alg:a1}
\footnotesize
\setstretch{0.95}
\begin{algorithmic}[1]

\Statex \hrulefill
\Statex \textbf{ASSUMPTION.} There exists a shared vocabulary $W$: a
        published list of keywords/phrases that every client draws on when
        describing what a class \emph{is}. Every client's description of the
        same real-world concept is therefore built from the same word pool,
        so identical concepts tend to yield identical or overlapping
        description strings.

\Statex \textbf{OBJECTS}
\Statex \quad $C_i$ \quad client $i$'s local set of classes (real ids).
\Statex \quad $\pi_i$ \quad client $i$'s private permutation
        $\mathrm{id} \mapsto \mathrm{masked\_id}$. Only masked ids ever leave
        client $i$.
\Statex \quad $W$ \quad the shared public vocabulary.
\Statex \quad $\mathrm{desc}_i[a] \subseteq W$ \quad the keyword(s) client $i$
        attaches to class $a$, describing it.

\Statex \hrulefill
\Statex \textbf{PHASE 1 --- build the local table. Local only; no
        communication.}
\For{each client $i$}
  \For{each class $a \in C_i$}
    \State $\mathrm{masked\_id} \gets \pi_i(a)$
    \State choose $\mathrm{desc}_i[a] \subseteq W$: the keywords that
           describe class $a$
    \State $\mathrm{Table}_i \gets \mathrm{Table}_i \cup
           \{(a,\ \mathrm{masked\_id},\ \mathrm{desc}_i[a])\}$
      \Comment{the local $|\mathrm{id}|\mathrm{masked\_id}|\mathrm{description}|$
               table}
  \EndFor
\EndFor

\Statex \hrulefill
\Statex \textbf{PHASE 2 --- PSI on descriptions. Between each client pair; no
        aggregator.}
\For{each client pair $(i,j)$, $i<j$}
  \State $E_{ij} \gets \emptyset$
  \State run $\psiexact$ between $\{\mathrm{desc}_i[a] : a \in C_i\}$ and
         $\{\mathrm{desc}_j[b] : b \in C_j\}$
    \Comment{plain PSI: reveals only which description strings are shared,
             not the descriptions of the non-matching classes}
  \For{each matched description $d$, held by class $a$ at $i$ and class $b$
       at $j$}
    \State $E_{ij} \gets E_{ij} \cup \{(\pi_i(a),\, \pi_j(b))\}$
      \Comment{a masked-id-to-masked-id relation -- never real ids}
  \EndFor
\EndFor
\State $E \gets \bigcup_{i<j} E_{ij}$
\State \Return $E$
  \Comment{this is what is handed to the aggregation server -- see
           Algorithm~\ref{alg:gid}}
\Statex \hrulefill

\end{algorithmic}
\end{algorithm}

\begin{algorithm}[p]
\caption{Fuzzy description PSI, no shared vocabulary (builds on
         Algorithm~\ref{alg:a1})}
\label{alg:a2}
\footnotesize
\setstretch{0.95}
\begin{algorithmic}[1]

\Statex \hrulefill
\Statex \textbf{WHAT CHANGES FROM ALGORITHM~\ref{alg:a1}.} There is no shared
        vocabulary any more: each client writes $\mathrm{desc}_i[a]$ in its
        own words. The same concept can then surface as different strings at
        different clients (``car'' at client $i$, ``automobile'' at client
        $j$), so exact-match PSI on the strings would miss the relation
        entirely. Matching is done on \emph{meaning}, not on the literal
        string: descriptions are embedded with a word-vector model
        (\emph{c\'i xi\`ang li\`ang}, word vectors) and compared with
        \emph{fuzzy} PSI, which reports a match whenever two embeddings lie
        within distance $\delta$ of one another rather than requiring
        equality.

\Statex \textbf{OBJECTS}
\Statex \quad $\mathrm{enc}_i$ \quad client $i$'s own word-vector /
        sentence-embedding model. May differ per client; never transmitted.
\Statex \quad $\mathrm{desc}_i[a]$ \quad free-text description of class $a$,
        in client $i$'s own vocabulary.
\Statex \quad $\Delta$ \quad threshold ladder, descending
        $0.90, 0.85, \ldots, 0.40$; the loosest rung is an absolute floor.
\Statex \quad $\pi_i$ \quad private permutation
        $\mathrm{id} \mapsto \mathrm{masked\_id}$, as in
        Algorithm~\ref{alg:a1}.

\Statex \hrulefill
\Statex \textbf{PHASE 1 --- build the local table. Local only; no
        communication.}
\For{each client $i$}
  \For{each class $a \in C_i$}
    \State $\mathrm{masked\_id} \gets \pi_i(a)$
    \State write $\mathrm{desc}_i[a]$, free text, own vocabulary
    \State $t_i[a] \gets \mathrm{enc}_i(\mathrm{desc}_i[a])$
      \Comment{the word-vector encoding of the description}
    \State $\mathrm{Table}_i \gets \mathrm{Table}_i \cup
           \{(a,\ \mathrm{masked\_id},\ \mathrm{desc}_i[a])\}$
  \EndFor
  \State $t_i \gets \unit(t_i)$
    \Comment{unit length so embeddings from different word-vector models are
             still compared by angle}
\EndFor

\Statex \hrulefill
\Statex \textbf{PHASE 2 --- fuzzy PSI on the embeddings. Between each client
        pair; no aggregator.}
\For{each client pair $(i,j)$, $i<j$}
  \State $R \gets \mathbf{0}$
  \For{$\delta \in \Delta$, tightest to loosest}
    \State $R \gets R + \fps(t_i,\, t_j,\, \delta)$
      \Comment{one execution; match BITS only -- summing the rungs turns
               yes/no answers into a rank that rises with similarity}
  \EndFor
  \State $M \gets \mutual(R)$
    \Comment{keep a pair only if each side is the other's top candidate}
  \State $E'_{ij} \gets \{(\pi_i(a),\, \pi_j(b)) : (a,b) \in M\}$
    \Comment{candidate relations -- description alone can still be fooled by
             coincidental near-synonyms}
\EndFor

\Statex \hrulefill
\Statex \textbf{PHASE 3 --- filter by image-set affinity: call
        Algorithm~\ref{alg:a4}.}
\For{each client pair $(i,j)$, $i<j$}
  \State $E_{ij} \gets \textsc{Algorithm4Filter}(E'_{ij},\, i,\, j)$
    \Comment{drops any pair that matched on wording but whose classes diverge
             in the image set -- e.g. ``car'' vs.\ ``cart''}
\EndFor
\State $E \gets \bigcup_{i<j} E_{ij}$
\State \Return $E$
  \Comment{handed to the aggregation server -- see Algorithm~\ref{alg:gid}}
\Statex \hrulefill

\end{algorithmic}
\end{algorithm}

\begin{algorithm}[p]
\caption{Affinity-generated description PSI, no description text at all
         (builds on Algorithm~\ref{alg:a2}, using affinity the same way as
         Algorithm~\ref{alg:a4})}
\label{alg:a3}
\footnotesize
\setstretch{0.95}
\begin{algorithmic}[1]

\Statex \hrulefill
\Statex \textbf{WHAT CHANGES FROM ALGORITHM~\ref{alg:a2}.} Rather than writing
        a free-text description and treating image-set affinity as a
        \emph{secondary filter} (Algorithm~\ref{alg:a4}, below), this variant
        drops text entirely: the ``description'' column of the local table is
        \emph{generated} the same way Algorithm~\ref{alg:a4} generates its
        filter signal -- an affinity profile over the public probe set -- and
        that profile is used directly, as the \emph{primary} matching signal.
        No shared vocabulary, no free text, no word-vector encoder is needed;
        classes with no meaningful name can still be bound.

\Statex \textbf{OBJECTS}
\Statex \quad $P$ \quad public probe set, as in Algorithm~\ref{alg:a4}.
\Statex \quad $\aff_i(a,p)$ \quad client $i$'s affinity criterion, as in
        Algorithm~\ref{alg:a4}.
\Statex \quad $\Delta$ \quad threshold ladder, as before.
\Statex \quad $\pi_i$ \quad private permutation
        $\mathrm{id} \mapsto \mathrm{masked\_id}$.

\Statex \hrulefill
\Statex \textbf{PHASE 1 --- build the local table, description generated by
        affinity. Local only; no communication.}
\Statex \emph{(An alternative way to fill this phase --- an attention-based
        projection onto the anchor set, rather than a raw affinity scan --- is
        given in Algorithm~\ref{alg:a3b}.)}
\For{each client $i$}
  \For{each class $a \in C_i$}
    \State $\mathrm{masked\_id} \gets \pi_i(a)$
    \State $S[a] \gets \big[\,\aff_i(a,p)\,\big]_{p \in P}$
      \Comment{the ``description'' -- generated, not written}
    \State $\mathrm{Table}_i \gets \mathrm{Table}_i \cup
           \{(a,\ \mathrm{masked\_id},\ S[a])\}$
  \EndFor
  \State $S \gets S - \mathrm{rowmean}(S)$
  \State $s_i \gets \unit(S)$
\EndFor

\Statex \hrulefill
\Statex \textbf{PHASE 2 --- fuzzy PSI directly on the affinity descriptions.
        Between each client pair; no aggregator.}
\For{each client pair $(i,j)$, $i<j$}
  \State $R \gets \mathbf{0}$
  \For{$\delta \in \Delta$, tightest to loosest}
    \State $R \gets R + \fps(s_i,\, s_j,\, \delta)$
  \EndFor
  \State $M \gets \mutual(R)$
    \Comment{same mutual-top-choice test as Algorithm~\ref{alg:a4} -- here it
             is the only check, since there is no second signal left to
             cross it against}
  \State $E_{ij} \gets \{(\pi_i(a),\, \pi_j(b)) : (a,b) \in M\}$
\EndFor
\State $E \gets \bigcup_{i<j} E_{ij}$
\State \Return $E$
  \Comment{handed to the aggregation server -- see Algorithm~\ref{alg:gid}}
\Statex \hrulefill

\end{algorithmic}
\end{algorithm}

\begin{algorithm}[p]
\caption{Attention-projected coordinates (alternate Phase 1 for
         Algorithm~\ref{alg:a3}: replaces the raw affinity scan with a
         learned projection onto a public anchor set)}
\label{alg:a3b}
\footnotesize
\setstretch{0.95}
\begin{algorithmic}[1]

\Statex \hrulefill
\Statex \textbf{WHAT CHANGES FROM ALGORITHM~\ref{alg:a3}.} Algorithm~\ref{alg:a3}
        scores class $a$ against every anchor independently
        ($\aff_i(a,p)$ per anchor $p$) and treats the resulting vector as the
        coordinate. Here, a single query vector $q_i(a)$ for class $a$ is
        instead \emph{attended} over a public bank of anchor keys/values
        $(K,V)$, so the coordinate is a learned, weighted pool over the anchor
        set rather than a per-anchor score list. This lets a small local
        model do the summarizing (which exemplars matter, how they combine)
        while the space being compared in stays fixed and public.

\Statex \textbf{OBJECTS}
\Statex \quad $A$ \quad the public anchor set (may reuse $P$).
\Statex \quad $K = \{k_p\}_{p\in A}$,\ $V=\{v_p\}_{p\in A}$ \quad
        \textbf{public} key/value embeddings of the anchors --- produced by
        one shared, published encoder. Identical for every client; this is
        what makes the resulting coordinates comparable at all.
\Statex \quad $\mathrm{enc}_i$ \quad client $i$'s own, private local model
        (may be a small LM) used only to summarize its exemplars of a class.
        Never transmitted.
\Statex \quad $\mathrm{proj}$ \quad a \textbf{public} projection head mapping
        a local summary into the same embedding space as $K,V$. Published,
        identical for all clients; only its input (client $i$'s private
        summary) differs per client.
\Statex \quad $\Delta$ \quad threshold ladder, as before.
\Statex \quad $\pi_i$ \quad private permutation
        $\mathrm{id} \mapsto \mathrm{masked\_id}$.

\Statex \hrulefill
\Statex \textbf{PHASE 1 --- build the local table, coordinates by attention.
        Local only; no communication.}
\For{each client $i$}
  \For{each class $a \in C_i$}
    \State $\mathrm{masked\_id} \gets \pi_i(a)$
    \State $r_i(a) \gets \mathrm{enc}_i(\text{exemplars or description of }a)$
      \Comment{private local summary of class $a$}
    \State $q_i(a) \gets \mathrm{proj}(r_i(a))$
      \Comment{projected into the shared, public embedding space}
    \State $(\alpha_i[a],\, \mathrm{coord}_i[a]) \gets
           \textsc{AttnCoord}(q_i(a),\, K,\, V)$
    \State $S[a] \gets \alpha_i[a]$ \textbf{or} $\mathrm{coord}_i[a]$
      \Comment{either output can serve as the coordinate -- attention weights
               (one entry per anchor, same shape as Algorithm~\ref{alg:a3}'s
               affinity vector) or the pooled embedding (dimension set by
               $v_p$)}
    \State $\mathrm{Table}_i \gets \mathrm{Table}_i \cup
           \{(a,\ \mathrm{masked\_id},\ S[a])\}$
  \EndFor
  \State $S \gets S - \mathrm{rowmean}(S)$
    \Comment{still applied even though $K,V$ are shared -- removes each
             client's own summarizer/exemplar-set bias}
  \State $s_i \gets \unit(S)$
\EndFor

\Statex \hrulefill
\Statex \textbf{PHASES 2 onward --- identical to Algorithm~\ref{alg:a3}:}
        fuzzy PSI on $s_i$ against $\Delta$, $\mutual(\cdot)$, edges returned
        to the aggregator (Algorithm~\ref{alg:gid}).
\Statex \hrulefill

\end{algorithmic}
\end{algorithm}

\begin{algorithm}[p]
\caption{Image-set affinity filter (subroutine called by Algorithm~\ref{alg:a2};
         the same mechanism used directly, as the primary signal, in
         Algorithm~\ref{alg:a3})}
\label{alg:a4}
\footnotesize
\setstretch{0.95}
\begin{algorithmic}[1]

\Statex \hrulefill
\Statex \textbf{PURPOSE.} Two classes can be described with near-synonymous
        words yet be visually/semantically different classes (or the reverse:
        worded very differently yet be the same class). This subroutine
        re-checks every candidate pair from the description signal against a
        second, independent signal built from each client's own image set,
        and keeps only the pairs both signals agree on.

\Statex \textbf{OBJECTS}
\Statex \quad $P$ \quad public probe set: a few hundred unlabelled public
        inputs, shared by all clients. Nobody's training data, no labels.
\Statex \quad $\aff_i(a,p)$ \quad client $i$'s affinity of class $a$ to public
        probe $p$ -- a classifier score, a $k$NN fraction over the local image
        set, a Bayes posterior, or any comparable criterion. The \emph{form}
        is agreed publicly; the model behind it is not shared.
\Statex \quad $\Delta$ \quad the same threshold ladder as
        Algorithm~\ref{alg:a2}.

\Statex \hrulefill
\Procedure{Algorithm4Filter}{$E'_{ij}$,\, $i$,\, $j$}
  \For{each class $a$ referenced in $E'_{ij}$ at client $i$}
    \State $S_i[a] \gets \big[\,\aff_i(a,p)\,\big]_{p \in P}$
      \Comment{one number per public probe -- the class's affinity profile}
  \EndFor
  \State $S_i \gets S_i - \mathrm{rowmean}(S_i)$
    \Comment{over $P$: drops the ``similar to everything'' bias, mandatory
             whenever $\aff \ge 0$}
  \State $s_i \gets \unit(S_i)$
  \State (client $j$ computes $s_j$ the same way)
  \State $R \gets \mathbf{0}$
  \For{$\delta \in \Delta$, tightest to loosest}
    \State $R \gets R + \fps(s_i,\, s_j,\, \delta)$
  \EndFor
  \State $M \gets \mutual(R)$
  \State \Return $\{(\pi_i(a),\pi_j(b)) \in E'_{ij} : (a,b) \in M\}$
    \Comment{only pairs where BOTH the description signal and the affinity
             signal agree survive}
\EndProcedure
\Statex \hrulefill

\end{algorithmic}
\end{algorithm}

\begin{algorithm}[t]
\caption{Global-id table and FL query flow (shared by all variants above)}
\label{alg:gid}
\small
\begin{algorithmic}[1]
\Require edge set $E$ (from Algorithm~\ref{alg:a1}, \ref{alg:a2}, \ref{alg:a3},
         or \ref{alg:a3b}), and every client's full LIST of masked ids

\Statex \textbf{AGGREGATOR --- build the global table.}
\State union--find: every $(i,\, \mathrm{masked\_id})$ starts in its own group
\For{$(i,\, \hat a,\, j,\, \hat b) \in E$}
  \If{the two groups already hold a masked id from the same client}
    \State \textbf{skip this edge}
      \Comment{one client's ids are distinct by construction, so this edge is
               the weak link of an inconsistent chain}
  \Else
    \State merge the two groups
  \EndIf
\EndFor
\State assign one $\mathrm{global\_id}$ per resulting group
  \Comment{a class that matched nothing simply owns its own global id -- every
           masked id must appear, or FL could not address that class}
\State \Return $\mathrm{GlobalTable} = \big|\,\mathrm{global\_id}\,\big|\,
       \mathrm{client\_id}\,\big|\,\mathrm{masked\_id}\,\big|$

\Statex \hrulefill
\Statex \textbf{FEDERATED LEARNING --- query flow, per round.}
\For{each global id $g$ the aggregator wants to train/aggregate}
  \For{each $(client\_id\ i,\ masked\_id\ \hat a)$ with $\mathrm{global\_id}=g$
       in $\mathrm{GlobalTable}$}
    \State aggregator sends $\hat a$ to client $i$
      \Comment{the aggregator only ever holds and sends masked ids}
    \State client $i$ looks up $a \gets \pi_i^{-1}(\hat a)$ locally
      \Comment{the real id never leaves the client}
    \State client $i$ returns the info needed for aggregation for class $a$
      (e.g.\ a local gradient, update, or statistic) tagged with $\hat a$
  \EndFor
  \State aggregator aggregates the returned contributions under $g$
\EndFor
\end{algorithmic}
\end{algorithm}
